-- Prompt Cache
local cachedPrompts = {}
local lastCacheUpdate = 0
local CACHE_REFRESH_INTERVAL = 2

-- ===================== BOULDER REVEAL =====================
task.spawn(function()
    local success, bf = pcall(function()
        return Workspace:WaitForChild("MountainDecorations"):WaitForChild("Boulders")
    end)
    if not success then return end
    
    for _, boulder in pairs(bf:GetChildren()) do
        for _, part in pairs(boulder:GetChildren()) do
            if part:IsA("MeshPart") then
                part.Transparency = 0
            end
        end
    end
    
    bf.DescendantAdded:Connect(function(descendant)
        if descendant:IsA("MeshPart") then
            descendant.Transparency = 0
        end
    end)
end)

-- ===================== PROMPT CACHE =====================
local function updatePromptCache()
    local now = tick()
    if now - lastCacheUpdate < CACHE_REFRESH_INTERVAL then return end
    
    cachedPrompts = {}
    for _, prompt in pairs(Workspace:GetDescendants()) do
        if prompt:IsA("ProximityPrompt") then
            table.insert(cachedPrompts, prompt)
        end
    end
    lastCacheUpdate = now
end

Workspace.DescendantAdded:Connect(function(descendant)
    if descendant:IsA("ProximityPrompt") then
        table.insert(cachedPrompts, descendant)
    elseif descendant:IsA("Folder") or descendant:IsA("Model") then
        lastCacheUpdate = 0
    end
end)

Workspace.DescendantRemoving:Connect(function(descendant)
    if descendant:IsA("ProximityPrompt") then
        for i, prompt in ipairs(cachedPrompts) do
            if prompt == descendant then
                table.remove(cachedPrompts, i)
                break
            end
        end
    end
end)

-- ===================== PROMPT SPAM =====================
local function spamPrompts()
    if not promptEnabled then return end
    
    local character = player.Character
    if not character then return end
    local root = character:FindFirstChild("HumanoidRootPart")
    if not root then return end
    
    local rootPos = root.Position
    updatePromptCache()
    
    local range = PROMPT_RANGE
    
    for _, prompt in ipairs(cachedPrompts) do
        if prompt.Enabled and prompt.Parent then
            local success, pos = pcall(function()
                return prompt.Parent.Position
            end)
            
            if success then
                local dist = (pos - rootPos).Magnitude
                if dist <= range then
                    pcall(fireproximityprompt, prompt)
                end
            end
        end
    end
end

local function startPromptSpam()
    if promptThread then return end
    
    promptThread = task.spawn(function()
        while promptEnabled do
            spamPrompts()
            task.wait(PROMPT_DELAY)
        end
        promptThread = nil
    end)
end

local function stopPromptSpam()
    promptEnabled = false
    promptThread = nil
end

-- ===================== TOUCH FEEDBACK =====================
local function addTouchFeedback(button)
    local originalColor = button.BackgroundColor3
    button.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.Touch or input.UserInputType == Enum.UserInputType.MouseButton1 then
            TweenService:Create(button, TweenInfo.new(0.1), {
                BackgroundColor3 = originalColor:Lerp(Color3.new(1, 1, 1), 0.3)
            }):Play()
        end
    end)
    button.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.Touch or input.UserInputType == Enum.UserInputType.MouseButton1 then
            TweenService:Create(button, TweenInfo.new(0.1), {
                BackgroundColor3 = originalColor
            }):Play()
        end
    end)
end

    if promptEnabled then
        fastPickupBtn.BackgroundColor3 = currentTheme.Accent
        fastPickupStroke.Color = currentTheme.Light
    else
        fastPickupBtn.BackgroundColor3 = Color3.fromRGB(40, 20, 60)
        fastPickupStroke.Color = Color3.fromRGB(130, 30, 180)
    end
end

-- Raycast
local raycastParams = RaycastParams.new()
raycastParams.FilterType = Enum.RaycastFilterType.Exclude

local function getMouseWorldPosition()
    raycastParams.FilterDescendantsInstances = {player.Character}
    
    local unitRay = camera:ScreenPointToRay(mouse.X, mouse.Y)
    local result = Workspace:Raycast(unitRay.Origin, unitRay.Direction * 2000, raycastParams)
    if result then return result.Position end
    return unitRay.Origin + unitRay.Direction * 40
end

local function getFrontPosition()
    local character = player.Character
    local root = character and character:FindFirstChild("HumanoidRootPart")
    if root then
        return root.Position + (camera.CFrame.LookVector * 9) + Vector3.new(0, 1.2, 0)
    end
    return camera.CFrame.Position + camera.CFrame.LookVector * 11
end

-- Dig Spam
local function startSpam(useFront)
    if connection then return end
    connection = RunService.Heartbeat:Connect(function()
        if not spamming then return end
        local pos = useFront and getFrontPosition() or getMouseWorldPosition()
        DigRequest:FireServer(currentPickaxe, pos)
    end)
    
    task.spawn(function()
        while spamming do
            task.wait(SPAM_DELAY)
        end
    end)
end

local function stopSpam()
    spamming = false
    if connection then
        connection:Disconnect()
        connection = nil
    end
end

local function toggleSpam(useFront)
    spamming = not spamming
    if spamming then
        startSpam(useFront)
    else
        stopSpam()
    end
    updateUI()
end

local function switchPickaxe()
    currentPickaxeIndex = currentPickaxeIndex % #pickaxes + 1
    currentPickaxe = pickaxes[currentPickaxeIndex]
    updateUI()
end

-- Prompt Toggle with Debounce
local promptDebounce = false
local function togglePrompt()
    if promptDebounce then return end
    promptDebounce = true
    
    promptEnabled = not promptEnabled
    
    if promptEnabled then
        startPromptSpam()
    else
        stopPromptSpam()
    end
    
    updateUI()
    task.delay(0.2, function() promptDebounce = false end)
end

-- Input handling
UserInputService.InputBegan:Connect(function(input, gp)
    if gp then return end
    
    if waitingForKey then
        if input.UserInputType == Enum.UserInputType.Keyboard then
            currentKey = input.KeyCode
            waitingForKey = false
            keyBtn.Text = "Change Dig Key"
            keyBtn.BackgroundColor3 = Color3.fromRGB(35, 16, 50)
            updateUI()
        end
        return
    end
    
    if waitingForPromptKey then
        if input.UserInputType == Enum.UserInputType.Keyboard then
            promptKey = input.KeyCode
            waitingForPromptKey = false
            promptKeyBtn.Text = "Pickup Key: " .. promptKey.Name
            promptKeyBtn.BackgroundColor3 = Color3.fromRGB(35, 16, 50)
            updateUI()
        end
        return
    end
    
    if input.KeyCode == currentKey then
        toggleSpam(false)
    elseif input.KeyCode == promptKey then
        if not promptDebounce then
            togglePrompt()
        end
    end
end)
